services/quote-service.js:347, :358 · routes/api/helpers/quotes-provider-routes.js:45-56

invoice.create แล้วตามด้วย invoiceLineItem.createMany โดยไม่อยู่ใน transaction ถ้าอันที่สองล้ม จะได้ใบกำกับภาษีที่มียอดรวมแต่ไม่มีรายการ — ซึ่ง ม.86/4 บังคับให้ต้องมี และไฟล์ export บัญชีก็อ่านจากรายการนั้น

และ Invoice.quoteId มี index แต่ไม่มี unique — กด "ออกใบแจ้งหนี้" สองครั้งจึงมินต์ใบกำกับภาษีสองใบจากใบเสนอราคาเดียวได้

วิธีแก้ — สองชั้น 
// 1) ทำให้เป็นอะตอม — ระบบมี $transaction ใช้อยู่แล้ว 122 จุด
await prisma.$transaction(async (tx) => {
  const invoice = await tx.invoice.create({ data: invoiceData });
  await tx.invoiceLineItem.createMany({ data: lineItems(invoice.id) });
  await tx.quote.update({ where: { id: quoteId }, data: { status: 'INVOICED' } });
});
-- 2) ทำให้กดซ้ำไม่ได้ ที่ระดับฐานข้อมูล
CREATE UNIQUE INDEX CONCURRENTLY "invoices_quoteId_key"
  ON "invoices" ("quoteId") WHERE "quoteId" IS NOT NULL;

CONCURRENTLY เพื่อไม่ล็อกตารางบนฐานที่ใช้งานอยู่ · WHERE ... IS NOT NULL เพื่อให้ใบที่ไม่ได้มาจากใบเสนอราคายังเป็น NULL ได้หลายใบ

services/quote-service.js:380-384

ข้อเสนอ — ระวังจุดที่ผู้ตรวจสอบเตือน

จำ personal-entity ไว้ชั่วคราวได้ เพราะเป็นค่าที่แทบไม่เปลี่ยนและไม่ใช่กลไกเพิกถอนสิทธิ์ แต่ผู้ตรวจสอบเตือนไว้ชัดว่า ห้าม cache ผลลบ — ผู้ใช้ที่เพิ่งถูกเพิ่มเข้า workspace จะถูกกันออกต่ออีกหนึ่งนาที 

const PERSONAL_TTL_MS = 60_000;
const cache = new Map(); // userId -> { id, exp }

async function findPersonalEntityCached(user) {
  const hit = cache.get(user.id);
  if (hit && hit.exp > Date.now()) return hit.id;
  const found = await findPersonalEntity(user);
  if (found) cache.set(user.id, { id: found, exp: Date.now() + PERSONAL_TTL_MS });
  return found;   // ไม่พบ = ไม่ cache
}

1,136 คอลัมน์เป็น text และไม่มี Postgres ENUM แม้แต่ตัวเดียว

คอลัมน์ที่เป็น enum โดยความหมาย — status, role, type — ล้วนเป็น text เปล่า ฐานข้อมูลจึงยอมรับค่าอะไรก็ได้ การพิมพ์ผิดหนึ่งตัวอักษรกลายเป็นสถานะใหม่ที่ไม่มีใครรู้จัก

เรียงตามความเสียหายต่อผู้ใช้หารด้วยแรงที่ต้องลง

  1. หยุดตัวกรอง input ไม่ให้ทำลายข้อความไทย

    นี่คือข้อเดียวในรายงานที่ทำข้อมูลเสียหายอยู่ตอนนี้ ทุกครั้งที่มีคนพิมพ์ -- หรือคำที่ขึ้นต้นด้วย on ตามด้วย = หรือหมายเหตุใน /* */ · และเสียแบบกู้ไม่ได้เพราะไม่มีใครรู้ว่าเสีย

    แก้ middleware หนึ่งไฟล์ · แต่ต้องตัดสินก่อนว่าจะปฏิเสธหรือปล่อยผ่าน

  2. สลับลำดับ code || error ในหน้าเว็บ

    บรรทัดเดียว ปลดล็อกข้อความไทยเฉพาะเรื่องที่เขียนไว้แล้วแต่ไม่เคยแสดง ใน 108 endpoint

    1 บรรทัด · ผู้ตรวจสอบยืนยันความปลอดภัยจาก 129 ค่า

  3. ปิดช่องออกใบกำกับภาษีซ้ำ (Tier C — operator merge)

    unique index บน quoteId + ห่อการมินต์ด้วย $transaction · ตอนนี้กดสองครั้งได้ใบภาษีสองใบ และใบที่ไม่มีรายการก็เกิดได้

    1 migration + 1 service · เส้นทางเงิน ผมเสนอ diff เท่านั้น

  4. เลิกใช้ count()+1 ตั้งเลขเอกสาร

    ใช้ตัวจ่ายเลขที่ระบบมีอยู่แล้ว · ตอนนี้ soft-delete หนึ่งใบทำให้ใบถัดไปชนแน่นอน

    แทนที่ฟังก์ชันเดียวด้วยตัวที่มีอยู่แล้ว

  5. ทำ OpenAPI ให้ซื่อสัตย์ เริ่มจากประตูสาธารณะ 17 ตัว

    ไม่ใช่ "เขียนให้ครบ 648" — เขียนเฉพาะผิวที่คนนอกต้องใช้ แล้วเพิ่ม probe บังคับ ไม่ให้เน่าอีก · และลบ operation เรื่องสลิปที่เลิกใช้แล้วออกก่อนเลย เพราะตอนนี้เอกสารสาธารณะโกหก

    งานเขียน + probe หนึ่งตัว

  6. ทดลอง relationJoins เป็นใบของตัวเอง

    231 include × หนึ่งรอบเดินทาง · บรรทัดเดียวในคอนฟิก แต่เปลี่ยน query ทั้งระบบ
