    function formatLastLogin(iso: string | null | undefined): string {
        if (!iso) return 'ยังไม่เคยล็อกอิน';
        try {
            const d = new Date(iso);
            return d.toLocaleString('th-TH', {
                year: 'numeric',
                month: 'short',
                day: '2-digit',
                hour: '2-digit',
                minute: '2-digit',
            });
        } catch {
            return iso;
        }
    }
